# 5. PCA Processing Function
def analyze_pca(df, title):
  X = StandardScaler().fit_transform(df)
  pca = PCA()
  scores = pca.fit_transform(X)

  scores_df = pd.DataFrame(scores, index=names, columns=["PC1", "PC2", "PC3"])
  loadings = pd.DataFrame(
      pca.components_.T, index=["Dec", "Jan", "Feb"], columns=["PC1", "PC2", "PC3"]
  )
  var_ratio = (pca.explained_variance_ratio_ * 100).round(1)

  print(f"--- {title} PCA RESULTS ---")
  print(f"Explained Variance Ratio (%): {var_ratio}")
  print("\nLoadings:")
  display(loadings.round(2))
  print("\n" + "=" * 40 + "\n")
  return scores_df, var_ratio


# Run PCA for both
scores_stagnant, var_stag = analyze_pca(df_stagnant, "STAGNANT WATER")
scores_flowing, var_flow = analyze_pca(df_flowing, "FLOWING WATER")

  for n in names:
    ax.annotate(
        n.replace("PCB-", ""),
        (scores_df.loc[n, "PC1"], scores_df.loc[n, "PC2"]),
        fontsize=8,
        alpha=0.85,
    )

  ax.axhline(0, color="grey", lw=0.5, linestyle="--")
  ax.axvline(0, color="grey", lw=0.5, linestyle="--")
  ax.set_xlabel(f"PC1 ({explained[0]}%)", fontsize=10)
  ax.set_ylabel(f"PC2 ({explained[1]}%)", fontsize=10)
  ax.set_title(title, fontsize=12, fontweight="bold")
